{
  var
    referent, /* [BP+00h] type=0x69 */
    InCrouch, /* [BP-01h] type=0x62 */
    Blastpac, /* [BP-03h] type=0x24 */
    target, /* [BP-05h] type=0x24 */
    item, /* [BP-07h] type=0x24 */
    counter, /* [BP-09h] type=0x69 */
    lastAnim, /* [BP-11h] type=0x69 */
    xOff, /* [BP-13h] type=0x69 */
    yOff, /* [BP-15h] type=0x69 */
    mine; /* [BP-17h] type=0x24 */

  set_info(0x0207, *(arg_06));
  process_exclude();
  Blastpac = *(arg_06);
  set_info(0x00f0, 0);
  lastAnim = Actor.getLastAnimSet(&global[0x003c]);
  Ultima8Engine.setAvatarInStasis(1);
  while (Actor.isBusy(&global[0x003c])) {
    spawn FREE.waitNTimerTicks(pid, 2, 0x00000000);
    suspend;
  }
  for item in nearby_items(shape=0x053a, origin=global[0x003c]) {
    target = item;
    FREE.slot_29(target, global[0x003c]);
    while ((retval > 3) && (counter != 6)) {
      spawn NPC.slot_22(Item.getDirFromTo16(pid, 1, Item.getY(target), Item.getX(Item.getY(Item.getX(target), &global[0x003c]), &global[0x003c])), &global[0x003c]);
      suspend;
      counter = (1 + counter);
    }
    suspend;
  }
  switch (Actor.getDir(&global[0x003c])) {
  case 0:
    xOff = -1;
    yOff = -3;
    break;
  case 1:
    xOff = 0;
    yOff = -3;
    break;
  case 2:
    xOff = 1;
    yOff = -3;
    break;
  case 3:
    xOff = 2;
    yOff = -2;
    break;
  case 4:
    xOff = 2;
    yOff = -1;
    break;
  case 5:
    xOff = 2;
    yOff = 0;
    break;
  case 6:
    xOff = 2;
    yOff = 2;
    break;
  case 7:
    xOff = 0;
    yOff = 2;
    break;
  case 8:
    xOff = -1;
    yOff = 2;
    break;
  case 9:
    xOff = -2;
    yOff = 2;
    break;
  case 10:
    xOff = -3;
    yOff = 2;
    break;
  case 11:
    xOff = -3;
    yOff = 1;
    break;
  case 12:
    xOff = -3;
    yOff = 0;
    break;
  case 13:
    xOff = -3;
    yOff = -1;
    break;
  case 14:
    xOff = -3;
    yOff = -2;
    break;
  case 15:
    xOff = -3;
    yOff = -3;
    break;
  }
  if (target) {
    if (Actor.getSomeFlagProbablyCrouch(&global[0x003c])) {
      InCrouch = 1;
    }
    else {
      InCrouch = 0;
      suspend;
    }
    Item.create(0, 0x053f, mine);
    Item.popToCoords((Item.getX((Item.getY(Item.getZ(&global[0x003c]), &global[0x003c]) + (yOff * 16)), &global[0x003c]) + (xOff * 16)), mine);
    spawn BLASTPAC.slot_20(mine, arg_06);
    while (Actor.isBusy(&global[0x003c])) {
      spawn FREE.waitNTimerTicks(pid, 2, 0x00000000);
      suspend;
    }
    spawn FREE.slot_21(pid, 13, 0x00000000);
    suspend;
    if (!InCrouch) {
      suspend;
      if (target) {
        suspend;
        suspend;
      }
    }
    else {
      spawn FREE.slot_21(pid, 0x00b0, 0x00000000);
      suspend;
      Item.create(0, BLAST_PAC, mine);
      Item.popToCoords(0, 0, 0, mine);
      MainActor.addItemCru(0, mine);
    }
  }
  else {
    spawn FREE.slot_21(pid, 0x00b0, 0x00000000);
    suspend;
    Item.create(0, BLAST_PAC, mine);
    Item.popToCoords(0, 0, 0, mine);
    MainActor.addItemCru(0, mine);
  }
  Ultima8Engine.setAvatarInStasis(0);
  return;

  /* post-return metadata (not executable):

    debug_symbol Blastpac [BP-03h] type=0x24 unk1=0x00 unk3=0x02
    debug_symbol target [BP-05h] type=0x24 unk1=0x00 unk3=0x02
    debug_symbol item [BP-07h] type=0x24 unk1=0x00 unk3=0x02
    debug_symbol counter [BP-09h] type=0x69 unk1=0x00 unk3=0x00
    debug_symbol lastAnim [BP-11h] type=0x69 unk1=0x00 unk3=0x00
    debug_symbol xOff [BP-13h] type=0x69 unk1=0x00 unk3=0x00
    debug_symbol yOff [BP-15h] type=0x69 unk1=0x00 unk3=0x00

  */
}
