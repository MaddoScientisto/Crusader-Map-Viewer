function hover_slot_21() /* entry=2576 class_id=0x0a0e slot=0x21 */
{
  var
    referent, /* [BP+00h] type=0x69 */
    ref, /* [BP+10h] type=0x69 */
    h, /* [BP-02h] type=0x69 */
    firstEgg, /* [BP-04h] type=0x69 */
    targetEgg, /* [BP-06h] type=0x69 */
    time, /* [BP-08h] type=0x69 */
    oldqhi, /* [BP-10h] type=0x69 */
    link, /* [BP-12h] type=0x69 */
    tx, /* [BP-14h] type=0x69 */
    ty, /* [BP-16h] type=0x69 */
    tz, /* [BP-17h] type=0x62 */
    hover, /* [BP-19h] type=0x24 */
    item; /* [BP-21h] type=0x24 */

  set_info(1, 0);
  hover = ref;
  h = Item.hurl(hover);
  oldqhi = h;
  firstEgg = (h & 15);
  targetEgg = firstEgg;
  while (1) {
    if ((Item.getStatus(hover) & 0x1000) == 0) {
      spawn HOVER.slot_23(pid, 1, ref, arg_06);
      suspend;
      return;
    }
    else {
      link = Item.unequip(hover);
      tx = -1;
      for item in nearby_items(shape=NUMBERS, origin=ref) {
        if (Item.getFrame(item) == targetEgg) {
          if ((link == 0) || (link == Item.unequip(item))) {
            tx = Item.getX(item);
            ty = Item.getY(item);
            tz = Item.getZ(item);
            time = Item.hurl(item);
          }
        }
      }
      if (tx == -1) {
        if (targetEgg == firstEgg) {
          /* bit_not */
          return;
        }
        targetEgg = firstEgg;
        h = (h & 0x00f0);
      }
      spawn HOVER.slot_20(pid, tz, ty, tx, ref, arg_06);
      suspend;
      /* dword_to_word */
      if (process_result == 1) {
        return;
      }
      else {
        spawn HOVER.slot_24(pid, targetEgg, ref, arg_06);
        suspend;
        if ((Item.getStatus(hover) & 0x1000) == 0) {
          spawn HOVER.slot_23(pid, 1, ref, arg_06);
          suspend;
          return;
        }
        else if (h & 0x0080) {
          spawn TRIGGER.slot_20(pid, (0 + 0x0080), item, arg_06);
          suspend;
        }
        if (time != 0) {
          spawn FREE.slot_32(pid, (time * 10), 0x00000000);
          suspend;
        }
        h = Item.hurl(hover);
        if ((h & 15) == targetEgg) {
          targetEgg = ((targetEgg + 1) & 15);
        }
        else {
          firstEgg = (h & 15);
          targetEgg = firstEgg;
          oldqhi = h;
        }
        h = (h & 0x00f0);
      }
    }
  }
  return;
}
